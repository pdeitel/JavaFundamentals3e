=============================
Compiling the AddressBook App
=============================
// macOS
javac --module-path $PATH_TO_FX --add-modules javafx.controls,javafx.fxml *.java

// Windows PowerShell
javac --module-path $env:PATH_TO_FX --add-modules javafx.controls,javafx.fxml *.java

// Windows Command Prompt
javac --module-path %PATH_TO_FX% --add-modules javafx.controls,javafx.fxml *.java

=============================
Running the AddressBook App
=============================
// macOS
java --enable-native-access=javafx.graphics,ALL-UNNAMED --module-path $PATH_TO_FX --add-modules javafx.controls,javafx.fxml -cp ".:../libs/sqlite-jdbc-3.54.1.0.jar" AddressBook

// Windows PowerShell
java --enable-native-access=javafx.graphics,ALL-UNNAMED --module-path $env:PATH_TO_FX --add-modules javafx.controls,javafx.fxml -cp ".;..\libs\sqlite-jdbc-3.54.1.0.jar" AddressBook

// Windows Command Prompt
java --enable-native-access=javafx.graphics,ALL-UNNAMED --module-path %PATH_TO_FX% --add-modules javafx.controls,javafx.fxml -cp ".;..\libs\sqlite-jdbc-3.54.1.0.jar" AddressBook
